-- =============================================
-- AUTO REDEEM CODE X2
-- =============================================

local function redeemCodes()
    print("[+] Auto redeeming codes...")
    
    local redeemed = 0
    local success = 0
    
    -- Tìm RemoteEvent để redeem code
    local remoteEvent = nil
    for _, v in pairs(game:GetService("ReplicatedStorage"):GetDescendants()) do
        if v.Name == "Code" or v.Name == "Redeem" or v.Name == "EnterCode" then
            remoteEvent = v
            break
        end
    end
    
    if not remoteEvent then
        print("[-] Cannot find redeem remote!")
        return
    end
    
    -- Redeem x2 EXP codes
    for _, code in pairs(CODES.x2EXP) do
        pcall(function()
            remoteEvent:FireServer(code)
            redeemed = redeemed + 1
            print("[+] Redeemed x2 EXP: " .. code)
            wait(0.5)
        end)
    end
    
    -- Redeem x2 Beli codes
    for _, code in pairs(CODES.x2Beli) do
        pcall(function()
            remoteEvent:FireServer(code)
            redeemed = redeemed + 1
            print("[+] Redeemed x2 Beli: " .. code)
            wait(0.5)
        end)
    end
    
    -- Redeem x2 Fragment codes
    for _, code in pairs(CODES.x2Fragment) do
        pcall(function()
            remoteEvent:FireServer(code)
            redeemed = redeemed + 1
            print("[+] Redeemed x2 Fragment: " .. code)
            wait(0.5)
        end)
    end
    
    -- Redeem General codes
    for _, code in pairs(CODES.General) do
        pcall(function()
            remoteEvent:FireServer(code)
            redeemed = redeemed + 1
            success = success + 1
            print("[+] Redeemed: " .. code)
            wait(0.3)
        end)
    end
    
    print("[+] Total redeemed: " .. redeemed .. " codes")
    return success
end

-- FIND MONSTERS
-- =============================================

local function getMonsters()
    local monsters = {}
    local rootPos = rootPart.Position
    
    local monsterNames = {
        "Bandit", "Brute", "Gorilla", "Pirate", 
        "Marine", "Sword Master", "Boss", "Chief",
        "Mob", "Thug", "Raider", "Viking"
    }
    
    for _, v in pairs(workspace:GetChildren()) do
        if v:IsA("Model") and v:FindFirstChild("Humanoid") and v:FindFirstChild("HumanoidRootPart") then
            if v.Name ~= player.Name and v.Humanoid.Health > 0 then
                local hrp = v:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local dist = (rootPos - hrp.Position).Magnitude
                    if dist < 300 then
                        table.insert(monsters, {
                            Model = v,
                            Humanoid = v.Humanoid,
                            RootPart = hrp,
                            Distance = dist,
                            Health = v.Humanoid.Health,
                            MaxHealth = v.Humanoid.MaxHealth,
                        })
                    end
                end
            end
        end
    end
    
    return monsters
end

local function getNearestMonster()
    local monsters = getMonsters()
    if #monsters == 0 then return nil end
    
    table.sort(monsters, function(a, b) return a.Distance < b.Distance end)
    return monsters[1]
end

-- =============================================
-- AUTO ATTACK
-- =============================================

local function attackMonster(monster)
    if not monster then return end
    
    local hrp = monster.RootPart
    if not hrp then return end
    
    rootPart.CFrame = hrp.CFrame * CFrame.new(0, 0, 5)
    wait(0.05)
    
    if KAITUN.AutoAttack then
        humanoid:Attack()
        wait(KAITUN.AttackSpeed)
        humanoid:Attack()
        wait(0.05)
    end
    
    if KAITUN.SpamSkills then
        for _, tool in pairs(character:GetChildren()) do
            if tool:IsA("Tool") then
                local remote = tool:FindFirstChild("RemoteEvent")
                if remote then
                    for _, key in pairs({"Z", "X", "C"}) do
                        remote:FireServer(key)
                        wait(0.05)
                    end
                end
            end
        end
    end
end

-- =============================================
-- AUTO HEAL
-- =============================================

local function autoHeal()
    if not KAITUN.AutoHeal then return end
    
    local hpPercent = (humanoid.Health / humanoid.MaxHealth) * 100
    if hpPercent < KAITUN.HealThreshold then
        for _, item in pairs(player.Backpack:GetChildren()) do
            if item:IsA("Tool") then
                local name = string.lower(item.Name)
                if name:find("heal") or name:find("potion") or name:find("food") then
                    humanoid:EquipTool(item)
                    wait(0.2)
                    item:Activate()
                    wait(0.5)
                    break
                end
            end
        end
    end
end

-- =============================================
-- AUTO COLLECT
-- =============================================

local function autoCollect()
    if not KAITUN.AutoCollect then return end
    
    local rootPos = rootPart.Position
    
    for _, v in pairs(workspace:GetChildren()) do
        if v:IsA("Tool") or v:IsA("Model") then
            local name = string.lower(v.Name)
            if name:find("drop") or name:find("beli") or name:find("fruit") or name:find("item") then
                local cd = v:FindFirstChild("ClickDetector")
                if cd then
                    local dist = (rootPos - v.Position).Magnitude
                    if dist < KAITUN.CollectRadius then
                        fireclickdetector(cd)
                        wait(0.1)
                    end
                end
            end
        end
    end
end

-- =============================================
-- AUTO QUEST
-- =============================================

local function getNearestQuest()
    local rootPos = rootPart.Position
    local nearest = nil
    local minDist = math.huge
    
    for _, v in pairs(workspace:GetChildren()) do
        if v:IsA("Model") and v:FindFirstChild("HumanoidRootPart") then
            if v.Name:find("Quest") or v.Name:find("NPC") or v.Name:find("Giver") then
                local hrp = v:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local dist = (rootPos - hrp.Position).Magnitude
                    if dist < minDist then
                        nearest = v
                        minDist = dist
                    end
                end
            end
        end
    end
    
    return nearest
end

local function autoQuest()
    if not KAITUN.AutoQuest then return end
    
    local quest = getNearestQuest()
    if not quest then return end
    
    local hrp = quest:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    
    local dist = (rootPart.Position - hrp.Position).Magnitude
    if dist > 20 then
        rootPart.CFrame = hrp.CFrame * CFrame.new(0, 0, 5)
        wait(0.5)
    end
    
    local remote = quest:FindFirstChild("RemoteEvent")
    if remote then
        remote:FireServer()
        wait(0.5)
    end
end

-- =============================================
-- MAIN LOOP
-- =============================================

coroutine.wrap(function()
    local lastHeal = 0
    local lastCollect = 0
    local lastQuest = 0
    local lastAFK = 0
    local lastRedeem = 0
    
    -- Auto redeem codes on start
    if KAITUN.AutoRedeem then
        wait(2)
        redeemCodes()
    end
    
    while KAITUN.AutoFarm and player:FindFirstChild("Humanoid") and humanoid.Health > 0 do
        local level = player:GetAttribute("Level") or 0
        
        if level >= KAITUN.TargetLevel then
            wait(10)
            continue
        end
        
        -- Speed
        humanoid.WalkSpeed = 25 * KAITUN.FarmSpeed
        humanoid.JumpPower = 55
        
        -- Find monster
        local monster = getNearestMonster()
        
        if monster then
            attackMonster(monster)
            
            if tick() - lastHeal > 5 then
                autoHeal()
                lastHeal = tick()
            end
        else
            local spawnPoints = {
                Vector3.new(100, 0, 100),
                Vector3.new(-100, 0, -100),
                Vector3.new(200, 0, -50),
                Vector3.new(-50, 0, 200)
            }
            rootPart.CFrame = CFrame.new(spawnPoints[math.random(1, #spawnPoints)])
            wait(0.5)
        end
        
        if tick() - lastCollect > 3 then
            autoCollect()
            lastCollect = tick()
        end
        
        if tick() - lastQuest > 30 then
            autoQuest()
            lastQuest = tick()
        end
        
        if tick() - lastAFK > 30 then
            pcall(function()
                virtualUser:CaptureController()
                virtualUser:ClickButton2(Vector2.new())
            end)
            lastAFK = tick()
        end
        
        -- Auto redeem every 5 minutes
        if tick() - lastRedeem > 300 then
            if KAITUN.AutoRedeem then
                redeemCodes()
            end
            lastRedeem = tick()
        end
        
        wait(0.05)
    end
end)()
